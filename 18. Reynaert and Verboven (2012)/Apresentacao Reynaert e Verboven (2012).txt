% SECAO 1: MOTIVACAO
%================================================

\begin{frame}{Motiva\c{c}\~{a}o e Problema de Pesquisa}
\begin{itemize}
  \item O modelo logit de coeficientes aleat\'{o}rios de Berry, Levinsohn e Pakes (1995) --- BLP --- \'{e} a principal ferramenta para estimar demanda por produtos diferenciados com dados agregados
  \item Captura heterogeneidade n\~{a}o observada nas prefer\^{e}ncias via coeficientes aleat\'{o}rios $\sigma^k$ sobre as caracter\'{i}sticas, gerando padr\~{o}es de substitui\c{c}\~{a}o flex\'{i}veis
  \item Apesar de avan\c{c}os recentes (MPEC, integra\c{c}\~{a}o polinomial, algoritmos aprimorados), o estimador GMM frequentemente apresenta vi\'{e}s, imprecis\~{a}o e instabilidade num\'{e}rica
\end{itemize}
\vspace{0.25cm}
\begin{block}{Hip\'{o}tese central}
Os problemas decorrem do uso de instrumentos ineficientes. Os instrumentos \'{o}timos de Chamberlain (1987) resolvem simultaneamente vi\'{e}s, efici\^{e}ncia e estabilidade do estimador GMM do modelo BLP.
\end{block}
\end{frame}

%================================================
% SECAO 2: REVISAO DA LITERATURA
%================================================

\begin{frame}{Revis\~{a}o da Literatura e Contribui\c{c}\~{a}o}
\textbf{Problemas documentados com o estimador BLP:}
\begin{itemize}
  \item \textbf{Knittel e Metaxoglou (2012):} converg\^{e}ncia global comprometida; estimativas muito sens\'{i}veis a valores iniciais
  \item \textbf{Dub\'{e}, Fox e Su (2012):} NFP propaga erros num\'{e}ricos do loop interno; prop\~{o}em MPEC como alternativa mais r\'{a}pida
  \item \textbf{Skrainka e Judd (2011):} integra\c{c}\~{a}o pMC imprecisa; integra\c{c}\~{a}o Sparse Grid \'{e} superior
  \item \textbf{Armstrong (2012):} instrumentos BLP s\~{a}o fracos para o pre\c{c}o; recomendam cost shifters
\end{itemize}
\end{frame}

\begin{frame}{Revis\~{a}o da Literatura e Contribui\c{c}\~{a}o (cont.)}
\begin{block}{Contribui\c{c}\~{a}o de Reynaert e Verboven (2012)}
\begin{itemize}
  \item Identificam o problema fundamental: instrumentos ineficientes para identificar $\sigma$ (vari\^{a}ncias dos coeficientes aleat\'{o}rios)
  \item Instrumentos \'{o}timos de Chamberlain (1987) resolvem vi\'{e}s, efici\^{e}ncia e estabilidade simultaneamente
  \item MPEC e integra\c{c}\~{a}o Sparse Grid s\'{o} produzem ganhos consistentes quando combinados com instrumentos \'{o}timos
\end{itemize}
\end{block}
\vspace{0.25cm}
\textbf{Compara\c{c}\~{a}o com outros trabalhos:}
\begin{itemize}
  \item Armstrong (2012) foca instrumentos para $\alpha$ (pre\c{c}o); Reynaert e Verboven focam instrumentos para $\sigma$ (heterogeneidade) --- contribui\c{c}\~{o}es complement\'{a}res
  \item Abordagem bayesiana (Jiang et al., 2009) tamb\'{e}m produz estimativas mais precisas, mas \'{e} mais lenta e exige suposic\~{o}es mais fortes
\end{itemize}
\end{frame}

%================================================
% SECAO 3: O MODELO
%================================================

\begin{frame}{O Modelo: Utilidade e Coeficientes Aleat\'{o}rios}
\textbf{Utilidade indireta condicional do consumidor $i$ pelo produto $j$ no mercado $t$:}
\begin{equation}
  u_{ijt} = x_{jt}\beta_i - \alpha p_{jt} + \xi_{jt} + \varepsilon_{ijt}
\end{equation}
\begin{itemize}
  \item $x_{jt}$: caracter\'{i}sticas observ\'{a}veis;\; $p_{jt}$: pre\c{c}o;\; $\xi_{jt}$: inobserv\'{a}vel de demanda
  \item $\alpha$: utilidade marginal da renda (homog\^{e}nea);\; $\varepsilon_{ijt}$: choque i.i.d.\ tipo I valor extremo
  \item Coeficiente aleat\'{o}rio: $\beta_i^k = \beta^k + \sigma^k \nu_i^k$, com $\nu_i^k \sim (0,1)$
\end{itemize}
\vspace{0.25cm}
\textbf{Decomposi\c{c}\~{a}o em utilidade m\'{e}dia e heterogeneidade individual:}
\begin{equation}
  u_{ijt} = \underbrace{\delta_{jt}}_{\text{utilidade m\'{e}dia}} + \underbrace{\mu_{jt}(\nu_i)}_{\text{heterog.\ individual}} + \varepsilon_{ijt}
\end{equation}
onde $\delta_{jt} \equiv x_{jt}\beta - \alpha p_{jt} + \xi_{jt}$ e $\mu_{jt}(\nu_i) \equiv \sum_k x_{jt}^k \sigma^k \nu_i^k$.
\end{frame}

\begin{frame}{O Modelo: Quotas de Mercado e Lado da Oferta}
\textbf{Quota de mercado predita (integral sobre a heterogeneidade dos consumidores):}
\begin{equation}
  s_{jt}(\delta_t, \sigma) = \int \frac{\exp(\delta_{jt} + \mu_{jt}(\nu))}{1 + \sum_{l=1}^J \exp(\delta_{lt} + \mu_{lt}(\nu))} \, dP_\nu(\nu)
\end{equation}
\begin{itemize}
  \item Integra\c{c}\~{a}o pseudo-Monte Carlo (pMC): $s_{jt} \approx \frac{1}{m}\sum_{i=1}^m [\cdots]$ --- r\'{a}pida mas imprecisa
  \item Integra\c{c}\~{a}o Sparse Grid: $s_{jt} \approx \sum_{i=1}^n \phi_i [\cdots]$ --- mais precisa e, com instrumentos \'{o}timos, mais r\'{a}pida
\end{itemize}
\vspace{0.25cm}
\textbf{Lado da oferta (competi\c{c}\~{a}o perfeita):}
\begin{equation}
  p_t = X_t \gamma_1 + W_t \gamma_2 + \omega_t
\end{equation}
Os cost shifters $W_t$ afetam apenas o custo --- servem como instrumentos para o pre\c{c}o end\'{o}geno.
\end{frame}

%================================================
% SECAO 4: GMM E IDENTIFICACAO
%================================================

\begin{frame}{O Estimador GMM e Estrat\'{e}gia de Identifica\c{c}\~{a}o}
\textbf{Restri\c{c}\~{a}o de momento central:}
\begin{equation}
  E[\xi_{jt} \mid X_t, w_{jt}] = 0 \;\Rightarrow\; E[\xi_{jt}\, z_{jt}] = 0
\end{equation}
\begin{itemize}
  \item $\xi_{jt}$ \'{e} independente em m\'{e}dia das caracter\'{i}sticas de todos os produtos e dos cost shifters
  \item GMM minimiza: $\min_\theta\; \xi(\theta)' Z A Z' \xi(\theta)$
\end{itemize}
\vspace{0.25cm}
\textbf{Dois algoritmos de estima\c{c}\~{a}o:}
\begin{itemize}
  \item \textbf{NFP (BLP):} loop externo minimiza GMM; loop interno inverte $s_t(\delta_t,\sigma)=s_t$ por mapeamento de contra\c{c}\~{a}o --- erros do loop interno propagam para o externo
  \item \textbf{MPEC (Dub\'{e} et al., 2012):} $\min_{\theta,\xi}\,\xi'ZAZ'\xi$ sujeito a $s(\delta,\sigma)=s$ --- elimina o loop interno; equivalente ao NFP em teoria, mais r\'{a}pido na pr\'{a}tica
\end{itemize}
\end{frame}

%================================================
% SECAO 5: INSTRUMENTOS OTIMOS
%================================================

\begin{frame}{Instrumentos \'{O}timos de Chamberlain (1987) --- Teoria}
\textbf{Tr\^{e}s conjuntos de instrumentos padr\~{a}o:}
\begin{itemize}
  \item $z^1_{jt} = (x_{jt}, w_{jt})$: caracter\'{i}sticas e cost shifters
  \item $z^2_{jt}$: $z^1$ + polin\^{o}mios (quadrados e intera\c{c}\~{o}es de $x_{jt}$ e $w_{jt}$)
  \item $z^3_{jt}$: $z^1$ + somas de caracter\'{i}sticas dos concorrentes $\sum_{k \neq j} x_{kt}$ (instrumentos BLP)
\end{itemize}
\vspace{0.25cm}
\textbf{Instrumentos \'{o}timos de Chamberlain (1987) --- m\'{a}xima efici\^{e}ncia assint\'{o}tica:}
\begin{equation}
  z_{jt} = E\!\left[\frac{\partial \xi_{jt}(\theta)}{\partial \theta'} \;\bigg|\; X_t, w_{jt}\right]
\end{equation}
Exploram a estrutura funcional do modelo para minimizar a vari\^{a}ncia assint\'{o}tica do estimador GMM.
\end{frame}

\begin{frame}{Instrumentos \'{O}timos --- Par\^{a}metros Lineares e N\~{a}o Lineares}
\textbf{Para par\^{a}metros lineares $\beta$ e $\alpha$ (interpreta\c{c}\~{a}o 2SLS):}
\begin{equation}
  E\!\left[\frac{\partial \xi_{jt}}{\partial \beta'}\right] = x_{jt}, \qquad
  E\!\left[\frac{\partial \xi_{jt}}{\partial \alpha}\right] = x_{jt}\gamma_1 + w_{jt}\gamma_2
\end{equation}
\begin{itemize}
  \item Instrumento para $\beta$: os pr\'{o}prios regressores $x_{jt}$
  \item Instrumento para $\alpha$: pre\c{c}o predito da 1\textordfeminine{} etapa (OLS sobre cost shifters) --- id\^{e}ntico ao 2SLS linear; n\~{a}o depende dos par\^{a}metros de demanda
\end{itemize}
\vspace{0.25cm}
\textbf{Para par\^{a}metros n\~{a}o lineares $\sigma$ (infeas\'{i}veis --- requerem 1\textordfeminine{} etapa):}
\begin{equation}
  E\!\left[\frac{\partial \xi_{jt}(\theta)}{\partial \sigma'} \;\bigg|\; X_t, w_{jt}\right] = E\!\left[\frac{\partial \delta_{jt}(s_t, \sigma)}{\partial \sigma'} \;\bigg|\; X_t, w_{jt}\right]
\end{equation}
Fun\c{c}\~{a}o n\~{a}o linear das caracter\'{i}sticas de todos os concorrentes; captura como $\delta_{jt}$ varia com a heterogeneidade $\sigma$.
\end{frame}

\begin{frame}{Implementa\c{c}\~{a}o dos Instrumentos \'{O}timos para $\sigma$}
\textbf{Implementa\c{c}\~{a}o aproximada ($z^4_{jt}$ --- BLP, 1999):}
\begin{enumerate}
  \item Obter estimativa inicial $\hat{\theta}=(\hat{\beta},\hat{\alpha},\hat{\sigma})$ usando $z^1$
  \item Calcular pre\c{c}o predito $\hat{p}_t = X_t\hat{\gamma}_1 + W_t\hat{\gamma}_2$
  \item Calcular $\hat{\delta}_t = X_t\hat{\beta} - \hat{\alpha}\hat{p}_t$ e quotas $\hat{s}_t = s_t(\hat{\delta}_t,\hat{\sigma})$
  \item Usar o Jacobiano $\partial\delta_t(\hat{s}_t,\sigma)/\partial\sigma'|_{\sigma=\hat{\sigma}}$ como instrumento
\end{enumerate}
\vspace{0.25cm}
\textbf{Implementa\c{c}\~{a}o exata ($z^5_{jt}$):} m\'{e}dia do Jacobiano sobre $K$ simula\c{c}\~{o}es de $\hat{\xi}_t$:
\begin{equation}
  \frac{1}{K}\sum_{k=1}^K \frac{\partial \delta_t(\hat{s}_t^k, \sigma)}{\partial \sigma'}\bigg|_{\sigma=\hat{\sigma}}
\end{equation}
\textbf{Resultado pr\'{a}tico crucial:} 1\textordfeminine{} etapa com logit simples ($\sigma=0$) e chute inicial para $\hat{\sigma}^1$ \'{e} suficiente --- evita estima\c{c}\~{a}o custosa do modelo completo de coeficientes aleat\'{o}rios.
\end{frame}

% SECAO 6: MONTE CARLO
%================================================

\begin{frame}{Processo Gerador de Dados e Detalhes Computacionais}
\textbf{Design Monte Carlo:} 1.000 conjuntos de dados; $T = 25$ mercados, $J = 10$ produtos.
\begin{itemize}
  \item $x_{jt} = (1, x^1_{jt})$, com $x^1_{jt} \sim U(1,2)$; cost shifters $w_{jt} \sim U(0,1)^3$
  \item $(\xi_{jt},\, \omega_{jt}) \sim N\!\left(\mathbf{0},\;\begin{bmatrix}1 & 0{,}7 \\ 0{,}7 & 1\end{bmatrix}\right)$
\end{itemize}
\vspace{0.25cm}
\textbf{Par\^{a}metros verdadeiros:} $\beta=(2,2)$, $\alpha=-2$, $\sigma=(0,1)$; $\gamma_2=(3,3,3)$ (instrumentos fortes para pre\c{c}o).
\vspace{0.25cm}
\textbf{Detalhes computacionais:}
\begin{itemize}
  \item Knitro 800 Interior$\backslash$Direct (Matlab); MPEC com Jacobiano e Hessiana anal\'{i}ticos
  \item 10 valores iniciais por dataset; seleciona menor valor da fun\c{c}\~{a}o objetivo
  \item Sparse Grid com 9 n\'{o}s (exata para polin\^{o}mios at\'{e} grau 17)
\end{itemize}
\end{frame}

\begin{frame}{Conclus\~{a}o}
\textbf{Os instrumentos \'{o}timos de Chamberlain (1987) resolvem os principais problemas do estimador GMM do modelo BLP:}
\begin{itemize}
  \item \textbf{Vi\'{e}s:} reduzido por fator $>10$ para $\sigma^1$ vs.\ melhor instrumento padr\~{a}o
  \item \textbf{Efici\^{e}ncia:} RMSE($\sigma^1$) cai de $\approx1{,}3$ para $\approx0{,}14$ --- redu\c{c}\~{a}o de $\approx90\%$
  \item \textbf{Estabilidade:} $\approx5\times$ menos varia\c{c}\~{a}o entre valores iniciais; zero n\~{a}o-converg\^{e}ncias
  \item \textbf{Melhoria com amostra:} RMSE decresce na ordem $\sqrt{J}$ e $\sqrt{T}$ (n\~{a}o ocorre com padr\~{a}o)
\end{itemize}
\vspace{0.25cm}
\textbf{Recomenda\c{c}\~{a}o pr\'{a}tica:} buscar cost shifters para identificar $\alpha$ (Armstrong, 2012) \textbf{e} aplicar instrumentos \'{o}timos de Chamberlain para identificar $\sigma$. 1\textordfeminine{} etapa simples (logit linear + chute para $\sigma$) \'{e} suficiente.
\end{frame}

\begin{frame}{Conclus\~{a}o (cont.) --- Avisos Cautelares}
\textbf{Avan\c{c}os computacionais s\'{o} mostram todo o potencial com instrumentos \'{o}timos:}
\begin{itemize}
  \item Com $z^4$: MPEC \'{e} $\approx10\times$ mais r\'{a}pido que NFP (vs.\ $1{,}75\times$ com $z^1$)
  \item Sparse Grid mais r\'{a}pida e precisa que pMC \textit{apenas} com instrumentos \'{o}timos
\end{itemize}
\vspace{0.25cm}
\textbf{Limita\c{c}\~{o}es e avisos:}
\begin{itemize}
  \item Resultados assumem $\xi_{jt}$ n\~{a}o correlacionado com caracter\'{i}sticas de \textit{todos} os produtos --- se violado, instrumentos \'{o}timos requerem modifica\c{c}\~{a}o (Crawford, 2011; Ackerberg e Crawford, 2009)
  \item Caracter\'{i}sticas dos produtos tratadas como ex\'{o}genas e n\~{a}o correlacionadas entre produtos
  \item Modelo bem especificado (distribui\c{c}\~{a}o normal para $\nu$) --- validade em dados reais \'{e} agenda de pesquisa aberta
\end{itemize}
\end{frame}
